Private Sub Form_Load()
    On Error GoTo ErrorHandler
    
    ' تهيئة وتعبئة عناصر الشاشة بالبيانات الافتراضية من جدول A
    Call LoadCompanyData
    Call LoadEmployeeDataDefaults
    Call LoadClientDataDefaults
    Call LoadItemDataDefaults
    Call LoadDimensionsDefaults
    
    Exit Sub
ErrorHandler:
    MsgBox "حدث خطأ أثناء تحميل شاشة التعريفات: " & Err.Description, vbCritical
End Sub
