Private Sub Form_Load()
    On Error GoTo ErrorHandler
    
    ' تحميل القيم الافتراضية والربط مع الجداول الأساسية لمردودات المشتريات
    Call LoadPurchaseReturnsDefaults
    
    Exit Sub
ErrorHandler:
    MsgBox "حدث خطأ أثناء تحميل شاشة مردودات المشتريات: " & Err.Description, vbCritical
End Sub

End Sub

Private Sub cmbSupplier_Change()
    ' عند اختيار المورد: يتم جلب البيانات المرتبطة تلقائياً (الحساب الفرعي، الرئيسي، السجل التجاري، البطاقة الضريبية، العنوان، التليفون)[cite: 14]
End Sub

Private Sub cmbItem_Change()
    ' عند اختيار الصنف الشرائي المرتجع: يتم جلب كود الصنف الشرائي، سعر الوحدة، والجرعات المرتبطة[cite: 14]
End Sub

Private Sub txtQuantityReturned_AfterUpdate()
    ' حساب إجمالي المرتجع تلقائياً = الكمية المرتجعة × سعر الوحدة[cite: 14]

End Sub

Private Sub cmdSavePurchaseReturns_Click()
    On Error GoTo ErrorHandler
    
    ' التحقق من اكتمال البيانات الأساسية (رقم المستند، المورد، الأصناف المرتجعة، الكميات، المخزن، مركز التكلفة، إلخ)[cite: 14]

ErrorHandler:

End Sub
