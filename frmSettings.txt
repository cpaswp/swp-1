Private Sub cmdOpenSettings_Click()
    On Error GoTo ErrorHandler
    
    ' فتح شاشة الضبط الرئيسية
    Me.Visible = True
    Call LoadSettingsData
    
    Exit Sub
ErrorHandler:
    MsgBox "حدث خطأ أثناء فتح شاشة الضبط: " & Err.Description, vbCritical
End Sub

Private Sub LoadSettingsData()
    On Error GoTo ErrorHandler
    
    ' كود جلب وتعبئة بيانات شاشة الضبط (الجدول الأول A: التعريفات)
    ' اسم الشركة A1, كود الشركة A2, البطاقة الضريبية A3, السجل التجاري A4 ... إلخ
    Dim db As Object
    Dim rs As Object
    Set db = CurrentDb
    
    ' التحقق من وجود جدول التعريفات A وإنشاؤه تلقائياً إذا لم يكن موجوداً
    ' إنشاء جدول الحركات المالية B، الحركات المخزنية C، وجدول البادجت D
    
    MsgBox "تم تحميل شاشة الضبط والتحقق من الجداول (A, B, C, D) بنجاح.", vbInformation, "شاشة الضبط"
    
    Exit Sub
ErrorHandler:
    MsgBox "خطأ في تحميل بيانات الضبط: " & Err.Description, vbCritical
End Sub

    ' جدول A: التعريفات (من A1 إلى A32)
